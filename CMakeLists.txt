cmake_minimum_required(VERSION 3.16)
project(arctag VERSION 0.3.0 LANGUAGES C)

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

# ONNX Runtime: prebuilt release unpacked by scripts/download_onnxruntime.sh into third_party/onnxruntime
set(ONNXRUNTIME_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/onnxruntime" CACHE PATH "ONNX Runtime prebuilt release dir (include/, lib/)")
find_path(ORT_INCLUDE onnxruntime_c_api.h PATHS "${ONNXRUNTIME_DIR}/include" "${ONNXRUNTIME_DIR}/include/onnxruntime" NO_DEFAULT_PATH)
find_library(ORT_LIB NAMES onnxruntime PATHS "${ONNXRUNTIME_DIR}/lib" NO_DEFAULT_PATH)
if(NOT ORT_INCLUDE OR NOT ORT_LIB)
  message(FATAL_ERROR "ONNX Runtime not found in ${ONNXRUNTIME_DIR}. Run scripts/download_onnxruntime.sh (or set -DONNXRUNTIME_DIR=...).")
endif()
message(STATUS "ONNX Runtime: ${ORT_LIB}")

# executables and the shared library find libonnxruntime next to the build (rpath) or via PATH on Windows
get_filename_component(ORT_LIB_DIR "${ORT_LIB}" DIRECTORY)
set(CMAKE_BUILD_RPATH "${ORT_LIB_DIR}")
set(CMAKE_INSTALL_RPATH "${ORT_LIB_DIR}")

option(ARCTAG_SHARED "build libarctag as a shared library" OFF)
option(ARCTAG_SANITIZE "build everything with AddressSanitizer + UndefinedBehaviorSanitizer (gcc / clang)" OFF)
if(ARCTAG_SANITIZE AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=address,undefined)
endif()
set(ARCTAG_SRC src/arctag.c src/tokenizer.c third_party/cjson/cJSON.c)
if(ARCTAG_SHARED)
  add_library(arctag SHARED ${ARCTAG_SRC})
  target_compile_definitions(arctag PRIVATE ARCTAG_BUILD_SHARED)
else()
  add_library(arctag STATIC ${ARCTAG_SRC})
endif()
# shared copy of the library for language bindings (the web demo loads it with Python ctypes): libarctag.so / .dylib, arctag.dll
option(ARCTAG_BINDINGS_LIB "also build libarctag as a shared library for language bindings (demo/)" ON)
set(ARCTAG_LIB_TARGETS arctag)
if(ARCTAG_BINDINGS_LIB AND NOT ARCTAG_SHARED)
  add_library(arctag_shared SHARED ${ARCTAG_SRC})
  target_compile_definitions(arctag_shared PRIVATE ARCTAG_BUILD_SHARED)
  set_target_properties(arctag_shared PROPERTIES OUTPUT_NAME arctag ARCHIVE_OUTPUT_NAME arctag_dll)
  list(APPEND ARCTAG_LIB_TARGETS arctag_shared)
endif()
foreach(lt ${ARCTAG_LIB_TARGETS})
  target_include_directories(${lt} PUBLIC include PRIVATE src third_party/cjson ${ORT_INCLUDE})
  target_link_libraries(${lt} PUBLIC ${ORT_LIB})
  if(NOT WIN32)
    target_link_libraries(${lt} PUBLIC m)
  endif()
  if(MSVC)
    target_compile_definitions(${lt} PRIVATE _CRT_SECURE_NO_WARNINGS)
  else()
    target_compile_options(${lt} PRIVATE -Wall -Wno-unused-function)
  endif()
endforeach()


add_executable(arctag_cli examples/arctag_cli.c)
target_link_libraries(arctag_cli arctag)
add_executable(arctag_server examples/arctag_server.c)
target_include_directories(arctag_server PRIVATE third_party/cjson)
target_link_libraries(arctag_server arctag)
if(WIN32)
  target_link_libraries(arctag_server ws2_32)
endif()
foreach(ex batch_tag stream_asr redact vertical_run compare_precision bench)
  add_executable(${ex} examples/${ex}.c)
  target_include_directories(${ex} PRIVATE third_party/cjson)
  target_link_libraries(${ex} arctag)
endforeach()
add_executable(test_tokenizer tests/test_tokenizer.c)
target_include_directories(test_tokenizer PRIVATE src third_party/cjson)
target_link_libraries(test_tokenizer arctag)
foreach(t test_parity test_config test_int8_agreement)
  add_executable(${t} tests/${t}.c)
  target_include_directories(${t} PRIVATE third_party/cjson)
  target_link_libraries(${t} arctag)
endforeach()

if(WIN32 AND EXISTS "${ORT_LIB_DIR}/onnxruntime.dll")
  foreach(t arctag_cli arctag_server batch_tag stream_asr redact vertical_run compare_precision bench test_tokenizer test_parity test_config test_int8_agreement)
    add_custom_command(TARGET ${t} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${ORT_LIB_DIR}/onnxruntime.dll" $<TARGET_FILE_DIR:${t}>)
  endforeach()
endif()

enable_testing()
set(ARCTAG_MODEL_DIR "${CMAKE_CURRENT_SOURCE_DIR}/model" CACHE PATH "model dir (scripts/fetch_model.sh)")
add_test(NAME tokenizer COMMAND test_tokenizer "${ARCTAG_MODEL_DIR}/tokenizer.json" "${CMAKE_CURRENT_SOURCE_DIR}/tests/expected/tokens.json")
# parity: scores within 1e-4 everywhere; ARCTAG_PARITY_EXACT=ON requires bit-equal scores (reference made with onnxruntime
# 1.20.1 on the same CPU kernels: holds on x86-64 Linux with the bundled ORT, not across CPU architectures)
option(ARCTAG_PARITY_EXACT "require bit-equal scores in the parity tests" OFF)
if(ARCTAG_PARITY_EXACT)
  set(PARITY_TOL 0)
else()
  set(PARITY_TOL 1e-4)
endif()
add_test(NAME parity_fp32 COMMAND test_parity "${ARCTAG_MODEL_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/expected/reference.json" --tol ${PARITY_TOL})
add_test(NAME parity_int8 COMMAND test_parity "${ARCTAG_MODEL_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/expected/reference_int8.json" --int8 --tol ${PARITY_TOL})
add_test(NAME int8_agreement COMMAND test_int8_agreement "${ARCTAG_MODEL_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/expected/reference.json")
add_test(NAME example_compare_precision COMMAND compare_precision -m "${ARCTAG_MODEL_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/examples/data/docs.txt")
set_tests_properties(parity_int8 int8_agreement example_compare_precision PROPERTIES SKIP_RETURN_CODE 77)
file(GLOB ARCTAG_CONFIGS "${CMAKE_CURRENT_SOURCE_DIR}/configs/*.json")
add_test(NAME config COMMAND test_config "${ARCTAG_MODEL_DIR}" ${ARCTAG_CONFIGS})
foreach(v contact_center meetings legal)
  add_test(NAME example_${v} COMMAND vertical_run "${CMAKE_CURRENT_SOURCE_DIR}/configs/${v}.json" -m "${ARCTAG_MODEL_DIR}"
           "${CMAKE_CURRENT_SOURCE_DIR}/examples/data/$<IF:$<STREQUAL:${v},contact_center>,call_center,$<IF:$<STREQUAL:${v},meetings>,meeting,court>>.txt")
endforeach()
add_test(NAME example_stream_asr COMMAND stream_asr -m "${ARCTAG_MODEL_DIR}" -c "${CMAKE_CURRENT_SOURCE_DIR}/configs/contact_center.json"
         "${CMAKE_CURRENT_SOURCE_DIR}/examples/data/asr_stream.jsonl")
add_test(NAME example_redact COMMAND redact -m "${ARCTAG_MODEL_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/examples/data/pii.txt")
add_test(NAME example_batch COMMAND batch_tag -m "${ARCTAG_MODEL_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/examples/data/docs.txt")
install(TARGETS ${ARCTAG_LIB_TARGETS} arctag_cli arctag_server batch_tag stream_asr redact vertical_run compare_precision bench)
install(FILES include/arctag.h DESTINATION include)
