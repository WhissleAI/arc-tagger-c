cmake_minimum_required(VERSION 3.16)
project(arctag VERSION 0.2.0 LANGUAGES C)

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

# The library is the native runtime: src/ and cJSON, nothing else. A universal macOS build needs no special flags:
#   cmake -S . -B build -DCMAKE_OSX_ARCHITECTURES="arm64;x86_64"
# MSVC has no build of it (it needs mmap, pthreads and GCC/Clang vector code): there the ONNX Runtime implementation below is the library.
option(ARCTAG_SHARED "build libarctag as a shared library" OFF)
option(ARCTAG_NO_ACCELERATE "Apple platforms: do not use Accelerate (for comparison; slower on long texts)" OFF)
option(ARCTAG_BUILD_ORT "also build arctag_ort, the ONNX Runtime implementation (ort/), for comparison" OFF)
if(MSVC)
  set(ARCTAG_NATIVE OFF)
  set(ARCTAG_BUILD_ORT ON)
  set(ARCTAG_MAIN arctag_ort)
else()
  set(ARCTAG_NATIVE ON)
  set(ARCTAG_MAIN arctag)
endif()

if(ARCTAG_NATIVE)
  set(ARCTAG_SRC src/arctag.c src/atn.c src/tokenizer.c third_party/cjson/cJSON.c)
  if(ARCTAG_SHARED)
    add_library(arctag SHARED ${ARCTAG_SRC})
  else()
    add_library(arctag STATIC ${ARCTAG_SRC})
  endif()
  target_include_directories(arctag PUBLIC include PRIVATE src third_party/cjson)
  target_compile_options(arctag PRIVATE -Wall -Wno-unused-function)
  find_package(Threads REQUIRED)
  target_link_libraries(arctag PUBLIC Threads::Threads m)
  if(ARCTAG_NO_ACCELERATE)
    target_compile_definitions(arctag PRIVATE ATN_NO_ACCELERATE)
  elseif(APPLE)
    target_link_libraries(arctag PUBLIC "-framework Accelerate")
  endif()
endif()

# The ONNX Runtime implementation: same API. Nothing above depends on it. scripts/download_onnxruntime.sh fetches the prebuilt library.
if(ARCTAG_BUILD_ORT)
  set(ONNXRUNTIME_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/onnxruntime" CACHE PATH "ONNX Runtime prebuilt release dir (include/, lib/)")
  find_path(ORT_INCLUDE onnxruntime_c_api.h PATHS "${ONNXRUNTIME_DIR}/include" "${ONNXRUNTIME_DIR}/include/onnxruntime" NO_DEFAULT_PATH)
  find_library(ORT_LIB NAMES onnxruntime PATHS "${ONNXRUNTIME_DIR}/lib" NO_DEFAULT_PATH)
  if(NOT ORT_INCLUDE OR NOT ORT_LIB)
    message(FATAL_ERROR "ONNX Runtime not found in ${ONNXRUNTIME_DIR}. Run scripts/download_onnxruntime.sh (or set -DONNXRUNTIME_DIR=...).")
  endif()
  if(MSVC AND ARCTAG_SHARED)
    add_library(arctag_ort SHARED ort/arctag_ort.c src/tokenizer.c third_party/cjson/cJSON.c)
    target_compile_definitions(arctag_ort PRIVATE ARCTAG_BUILD_SHARED)
  else()
    add_library(arctag_ort STATIC ort/arctag_ort.c src/tokenizer.c third_party/cjson/cJSON.c)
  endif()
  target_include_directories(arctag_ort PUBLIC include PRIVATE src third_party/cjson ${ORT_INCLUDE})
  target_link_libraries(arctag_ort PUBLIC ${ORT_LIB})
  if(MSVC)
    target_compile_definitions(arctag_ort PRIVATE _CRT_SECURE_NO_WARNINGS)
  else()
    target_link_libraries(arctag_ort PUBLIC m)
  endif()
  get_filename_component(ORT_LIB_DIR "${ORT_LIB}" DIRECTORY)
  set(CMAKE_BUILD_RPATH "${ORT_LIB_DIR}")
endif()

add_executable(arctag_cli examples/arctag_cli.c)
target_link_libraries(arctag_cli ${ARCTAG_MAIN})
add_executable(arctag_server examples/arctag_server.c)
target_include_directories(arctag_server PRIVATE third_party/cjson)
target_link_libraries(arctag_server ${ARCTAG_MAIN})
if(WIN32)
  target_link_libraries(arctag_server ws2_32)
endif()
add_executable(test_tokenizer tests/test_tokenizer.c)
target_include_directories(test_tokenizer PRIVATE src third_party/cjson)
target_link_libraries(test_tokenizer ${ARCTAG_MAIN})
add_executable(test_parity tests/test_parity.c)
target_include_directories(test_parity PRIVATE third_party/cjson)
target_link_libraries(test_parity ${ARCTAG_MAIN})
if(ARCTAG_NATIVE AND ARCTAG_BUILD_ORT)
  add_executable(test_parity_ort tests/test_parity.c)
  target_include_directories(test_parity_ort PRIVATE third_party/cjson)
  target_link_libraries(test_parity_ort arctag_ort)
endif()
if(WIN32 AND EXISTS "${ORT_LIB_DIR}/onnxruntime.dll")
  foreach(t arctag_cli arctag_server test_tokenizer test_parity)
    add_custom_command(TARGET ${t} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${ORT_LIB_DIR}/onnxruntime.dll" $<TARGET_FILE_DIR:${t}>)
  endforeach()
endif()

enable_testing()
set(ARCTAG_MODEL_DIR "${CMAKE_CURRENT_SOURCE_DIR}/model" CACHE PATH "model dir (scripts/fetch_model.sh, then tools/export_model.py)")
add_test(NAME tokenizer COMMAND test_tokenizer "${ARCTAG_MODEL_DIR}/tokenizer.json" "${CMAKE_CURRENT_SOURCE_DIR}/tests/expected/tokens.json")
add_test(NAME parity COMMAND test_parity "${ARCTAG_MODEL_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/expected/reference.json")
if(ARCTAG_NATIVE)
  add_executable(test_stream tests/test_stream.c)
  target_include_directories(test_stream PRIVATE third_party/cjson)
  target_link_libraries(test_stream arctag)
  add_test(NAME stream COMMAND test_stream "${ARCTAG_MODEL_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/expected/reference.json")
endif()
install(TARGETS ${ARCTAG_MAIN} arctag_cli arctag_server)
install(FILES include/arctag.h DESTINATION include)
